### 3.1 导入引擎 (Parser)

#### 3.1.1 解析器接口

```typescript
// src/parsers/index.ts
interface Parser<T = any> {
  name: string;
  extensions: string[];           // 支持的文件扩展名
  mimeTypes: string[];            // 支持的 MIME 类型
  
  // 解析文件
  parse(file: File): Promise<ParseResult<T>>;
  
  // 检测是否能解析（预览前 1KB 内容）
  canParse(sample: string): boolean;
}

interface ParseResult<T> {
  format: string;                 // 解析格式标识
  rawData: T;                     // 原始解析数据
  messages: RawMessage[];         // 标准化后的消息
  participants: RawParticipant[]; // 参与者信息
  metadata: ParseMetadata;        // 元数据
}

interface RawMessage {
  id: string;
  timestamp: number;
  senderName: string;             // 显示昵称
  senderWxid?: string;            // 微信 ID（可能缺失）
  content: string;
  type: 'text' | 'image' | 'voice' | 'video' | 'file' | 'system';
}

interface RawParticipant {
  name: string;
  wxid?: string;
  avatar?: string;
  isSystem?: boolean;
}
```

#### 3.1.2 解析流程

```
[用户拖拽文件]
     ↓
[读取文件前 1KB]
     ↓
[遍历注册器，调用 canParse() 匹配]
     ↓
[匹配成功 → 调用 parse() 完整解析]
     ↓
[标准化为 RawMessage[] + RawParticipant[]]
     ↓
[展示预览（前 20 条消息）]
     ↓
[等待用户确认身份]
     ↓
[写入数据库]
```

### 3.2 身份识别系统 (Identity)

#### 3.2.1 核心逻辑

```typescript
// src/stores/identity.ts
interface IdentityState {
  // 用户确认的身份映射
  confirmedMapping: Map<string, 'self' | 'other'>;  // name/wxid -> role
  
  // 曾用名记录
  aliasHistory: Alias[];
  
  // 自动推断的结果（供参考）
  inferredSelf: string[];
}

// 核心函数
function resolveSenderIdentity(
  msg: Message,
  state: IdentityState
): 'self' | 'other' | 'unknown' {
  // 1. 优先用 wxid 匹配
  if (msg.senderWxid && state.confirmedMapping.has(msg.senderWxid)) {
    return state.confirmedMapping.get(msg.senderWxid)!;
  }
  
  // 2. 用当前昵称匹配
  if (state.confirmedMapping.has(msg.senderName)) {
    return state.confirmedMapping.get(msg.senderName)!;
  }
  
  // 3. 用曾用名时间线匹配
  for (const alias of state.aliasHistory) {
    if (alias.name === msg.senderName && 
        msg.timestamp >= alias.startTime &&
        (!alias.endTime || msg.timestamp <= alias.endTime)) {
      return state.confirmedMapping.get('self')!;
    }
  }
  
  return 'unknown';
}
```

import Dexie, { Table } from 'dexie';

export interface DbMessage {
  id: string;                     // 唯一 ID
  timestamp: number;
  senderId: string;               // 内部参与者 ID
  content: string;
  type: MessageType;
  emotion?: EmotionLabel;         // 预计算的情绪标签
  wordCount: number;
  isSelf: boolean;                // 是否用户本人
  sessionId: string;              // 所属会话
}

export interface DbParticipant {
  id: string;                     // 内部 ID
  wxid?: string;
  names: string[];                // 所有出现过的名称
  role: 'self' | 'other';
  messageCount: number;
  firstSeen: number;
  lastSeen: number;
}

export interface DbSession {
  id: string;
  name: string;                   // 会话名称（对方昵称/群名）
  type: 'private' | 'group';
  importedAt: number;
  messageCount: number;
  timeRange: [number, number];    // [first, last]
}

export class ChatMindDB extends Dexie {
  messages!: Table<DbMessage>;
  participants!: Table<DbParticipant>;
  sessions!: Table<DbSession>;
  
  constructor() {
    super('ChatMindDB');
    this.version(1).stores({
      messages: 'id, [sessionId+timestamp], senderId, timestamp, emotion',
      participants: 'id, names, role',
      sessions: 'id, importedAt',
    });
  }
}
```

#### 3.3.2 索引策略

```
messages 表:
  - 主键: id
  - 复合索引: [sessionId + timestamp] → 按会话分页查询
  - 单索引: senderId → 统计某人发言
  - 单索引: timestamp → 时间范围查询
  - 单索引: emotion → 情绪筛选

function calculateStatistics(messages: DbMessage[]): StatisticsResult {

      delays.push(curr.timestamp - prev.timestamp);

type EmotionLabel = 'positive' | 'negative' | 'neutral' | 'angry' | 'sad' | 'affectionate';

interface EmotionResult {
  label: EmotionLabel;
  score: number;                  // 置信度 0-1
  method: 'rule' | 'ai';          // 分析方法
}

// 规则层：快速本地分析
function ruleBasedEmotion(text: string): EmotionResult | null {
  // 正面词库
  const positiveWords = ['开心', '喜欢', '爱你', '谢谢', '哈哈', '棒', '好'];
  // 负面词库
  const negativeWords = ['烦', '讨厌', '生气', '失望', '难过', '滚', '无语'];
  // 愤怒词库
  const angryWords = ['滚', '操', '他妈', '有病', '傻逼', '恶心'];
  
  let score = 0;
  let matched = false;
  
  for (const w of positiveWords) {
    if (text.includes(w)) { score += 1; matched = true; }
  }
  for (const w of negativeWords) {
    if (text.includes(w)) { score -= 1; matched = true; }
  }
  for (const w of angryWords) {
    if (text.includes(w)) { return { label: 'angry', score: 0.9, method: 'rule' }; }
  }
  
  if (!matched) return null;
  
  return {
    label: score > 0 ? 'positive' : score < 0 ? 'negative' : 'neutral',
    score: Math.min(Math.abs(score) * 0.3 + 0.5, 0.9),
    method: 'rule',
  };
}

### 4.1 大数据处理

| 场景 | 策略 | 实现 |

// 脱敏函数：调用云端 AI 前清理敏感信息
function sanitizeForAI(messages: DbMessage[]): SanitizedMessage[] {
  return messages.map(m => ({
    ...m,
    // 替换具体昵称为 role 标识
